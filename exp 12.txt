
def display_board(board):
    print()
    for i in range(3):
        print(" | ".join(board[i * 3:i * 3 + 3]))
        if i < 2:
            print("--+---+--")
    print()


def check_winner(board, player):
    win_positions = [
        (0, 1, 2), (3, 4, 5), (6, 7, 8),
        (0, 3, 6), (1, 4, 7), (2, 5, 8),
        (0, 4, 8), (2, 4, 6)
    ]

    for a, b, c in win_positions:
        if board[a] == board[b] == board[c] == player:
            return True
    return False


board = [" "] * 9
player = "X"

print("TIC-TAC-TOE")
print("Positions:")
print("1 | 2 | 3")
print("--+---+--")
print("4 | 5 | 6")
print("--+---+--")
print("7 | 8 | 9")

while True:
    display_board(board)
    position = int(input(f"Player {player}, enter position (1-9): "))

    if position < 1 or position > 9 or board[position - 1] != " ":
        print("Invalid position! Try again.")
        continue

    board[position - 1] = player

    if check_winner(board, player):
        display_board(board)
        print("Player", player, "wins!")
        break

    if " " not in board:
        display_board(board)
        print("Game Draw!")
        break

    player = "O" if player == "X" else "X"
