def print_board(board, n):
    print("\nSolution:")
    
    for i in range(n):
        for j in range(n):
            if board[i] == j:
                print("Q", end=" ")
            else:
                print(".", end=" ")
        print()


def is_safe(board, row, col):
    
    # Check previous rows
    for i in range(row):
        
        # Same column
        if board[i] == col:
            return False
        
        # Same diagonal
        if abs(board[i] - col) == abs(i - row):
            return False

    return True


def solve(board, row, n):
    
    # All queens are placed
    if row == n:
        return True

    # Try each column
    for col in range(n):
        
        if is_safe(board, row, col):
            
            # Place queen
            board[row] = col

            # Move to next row
            if solve(board, row + 1, n):
                return True

            # Backtrack
            board[row] = -1

    return False


# Get input from user
n = int(input("Enter the number of queens: "))

if n <= 0:
    print("Enter a positive number.")

else:
    board = [-1] * n

    if solve(board, 0, n):
        print_board(board, n)
    else:
        print("No solution exists.")
