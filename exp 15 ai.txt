
def minimax(board, depth, alpha, beta, isMax):
    if "O" in board and winner(board, "O"):
        return 10 - depth
    if winner(board, "X"):
        return depth - 10
    if " " not in board:
        return 0

    if isMax:
        best = -999
        for i in range(9):
            if board[i] == " ":
                board[i] = "O"
                best = max(best, minimax(board, depth+1,
                                         alpha, beta, False))
                board[i] = " "
                alpha = max(alpha, best)
                if beta <= alpha:
                    break
        return best
    else:
        best = 999
        for i in range(9):
            if board[i] == " ":
                board[i] = "X"
                best = min(best, minimax(board, depth+1,
                                         alpha, beta, True))
                board[i] = " "
                beta = min(beta, best)
                if beta <= alpha:
                    break
        return best


def winner(b, p):
    lines = [(0,1,2),(3,4,5),(6,7,8),
             (0,3,6),(1,4,7),(2,5,8),
             (0,4,8),(2,4,6)]
    return any(all(b[i] == p for i in line) for line in lines)


def show(b):
    for i in range(0, 9, 3):
        print("|".join(b[i:i+3]))
        if i < 6:
            print("-----")


board = [" "] * 9
print("Positions: 1 to 9")
show([str(i) for i in range(1, 10)])

while True:
    move = int(input("Your move (1-9): ")) - 1
    if move < 0 or move > 8 or board[move] != " ":
        print("Invalid move!")
        continue

    board[move] = "X"

    if winner(board, "X"):
        show(board)
        print("You win!")
        break
    if " " not in board:
        show(board)
        print("Draw!")
        break

    best, move = -999, -1
    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            score = minimax(board, 0, -999, 999, False)
            board[i] = " "
            if score > best:
                best, move = score, i

    board[move] = "O"
    print("Computer's move:")
    show(board)

    if winner(board, "O"):
        print("Computer wins!")
        break
    if " " not in board:
        print("Draw!")
        break
